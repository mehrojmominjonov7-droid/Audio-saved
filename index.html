"""
main.py — botni ishga tushiruvchi asosiy fayl.

Ishga tushirish:  python main.py
"""

import asyncio
import logging
import sys

from aiogram import Bot, Dispatcher
from aiogram.client.default import DefaultBotProperties
from aiogram.enums import ParseMode

from config import BOT_TOKEN
from downloader import cleanup_all_downloads
from handlers import router

logger = logging.getLogger(__name__)


async def main() -> None:
    # Token mavjudligini tekshiramiz
    if not BOT_TOKEN:
        logger.critical("BOT_TOKEN topilmadi! .env fayliga BOT_TOKEN=... qo'shing.")
        sys.exit(1)

    # Oldingi ishdan qolgan vaqtinchalik fayllarni tozalaymiz
    cleanup_all_downloads()

    # Bot obyekti (xabarlar standart holda HTML formatida yuboriladi)
    bot = Bot(
        token=BOT_TOKEN,
        default=DefaultBotProperties(parse_mode=ParseMode.HTML),
    )

    dp = Dispatcher()
    dp.include_router(router)

    try:
        me = await bot.get_me()
        logger.info("Bot ishga tushdi: @%s", me.username)

        # Bot o'chiq paytida to'planib qolgan eski xabarlarni o'tkazib yuboramiz
        await bot.delete_webhook(drop_pending_updates=True)
        await dp.start_polling(bot)
    finally:
        # Bot to'xtaganda sessiyani yopamiz va fayllarni tozalaymiz
        await bot.session.close()
        cleanup_all_downloads()
        logger.info("Bot to'xtatildi.")


if __name__ == "__main__":
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s | %(levelname)s | %(name)s | %(message)s",
    )
    try:
        asyncio.run(main())
    except (KeyboardInterrupt, SystemExit):
        logger.info("Bot qo'lda to'xtatildi.")

import asyncio
import html
import logging
import re
import uuid
from collections import OrderedDict
from typing import Optional

from aiogram import Bot, F, Router
from aiogram.enums import ChatAction
from aiogram.filters import CommandStart
from aiogram.types import (
    CallbackQuery,
    FSInputFile,
    InlineKeyboardButton,
    InlineKeyboardMarkup,
    Message,
)

from config import (
    AUDIO_BUTTON_TEXT,
    AUDIO_DOWNLOADING_TEXT,
    DOWNLOADING_TEXT,
    ERROR_TEXT,
    EXPIRED_LINK_TEXT,
    MAX_CONCURRENT_DOWNLOADS,
    NOT_A_LINK_TEXT,
    START_TEXT,
    TOO_LARGE_TEXT,
    UNSUPPORTED_LINK_TEXT,
)
from downloader import (
    FileTooLargeError,
    cleanup_folder,
    detect_platform,
    download_audio,
    download_video,
    extract_url,
)

logger = logging.getLogger(__name__)

router = Router()

# Bir vaqtda ishlayotgan yuklashlar sonini cheklaymiz
DOWNLOAD_SEMAPHORE = asyncio.Semaphore(MAX_CONCURRENT_DOWNLOADS)

# Callback tugmasi ma'lumoti 64 baytdan oshmasligi kerak, shuning uchun uzun
# havolani tugmaga sig'dirib bo'lmaydi. Havolani xotirada qisqa kalit bilan saqlaymiz.
URL_CACHE: "OrderedDict[str, str]" = OrderedDict()
URL_CACHE_LIMIT = 1000  # eng ko'pi bilan nechta havola saqlanadi


# ---------------------------------------------------------------------------
# Yordamchi funksiyalar
# ---------------------------------------------------------------------------


def remember_url(url: str) -> str:
    """Havolani keshga saqlab, qisqa kalit qaytaradi."""
    key = uuid.uuid4().hex[:12]
    URL_CACHE[key] = url
    # Kesh to'lib ketsa, eng eskilarini o'chiramiz
    while len(URL_CACHE) > URL_CACHE_LIMIT:
        URL_CACHE.popitem(last=False)
    return key


def audio_keyboard(url_key: str) -> InlineKeyboardMarkup:
    """Video ostidagi "Musiqasini ajratib olish" tugmasi."""
    return InlineKeyboardMarkup(
        inline_keyboard=[
            [
                InlineKeyboardButton(
                    text=AUDIO_BUTTON_TEXT,
                    callback_data=f"audio:{url_key}",
                )
            ]
        ]
    )


async def safe_delete(message: Optional[Message]) -> None:
    """Xabarni o'chirishga urinadi; xatolik bo'lsa e'tiborsiz qoldiradi."""
    if not message:
        return
    try:
        await message.delete()
    except Exception as exc:  # noqa: BLE001
        logger.debug("Xabarni o'chirib bo'lmadi: %s", exc)


def safe_filename(name: str, default: str = "audio") -> str:
    """Fayl nomidan taqiqlangan belgilarni olib tashlaydi."""
    cleaned = re.sub(r'[\\/:*?"<>|\r\n]+', " ", name).strip()
    return cleaned[:60] or default

@router.message(CommandStart())
async def cmd_start(message: Message) -> None:
    """Xush kelibsiz xabarini yuboradi."""
    try:
        await message.answer(START_TEXT)
    except Exception as exc:  # noqa: BLE001
        logger.exception("/start xatosi: %s", exc)

@router.message(F.text)
async def handle_link(message: Message, bot: Bot) -> None:
    """Foydalanuvchi yuborgan havolani qabul qilib, videoni yuklab beradi."""
    url = extract_url(message.text or "")

    # 1) Xabarda havola bormi?
    if not url:
        await message.answer(NOT_A_LINK_TEXT)
        return

    # 2) Havola qo'llab-quvvatlanadigan tarmoqqa tegishlimi?
    platform = detect_platform(url)
    if not platform:
        await message.answer(UNSUPPORTED_LINK_TEXT)
        return

    logger.info("Yangi so'rov: user=%s platform=%s", message.from_user.id if message.from_user else "?", platform)

    # 3) Holat xabari
    status_msg: Optional[Message] = None
    result = None

    try:
        status_msg = await message.answer(DOWNLOADING_TEXT)
        await bot.send_chat_action(message.chat.id, ChatAction.UPLOAD_VIDEO)

        # 4) Yuklab olish (bir vaqtda cheklangan miqdorda)
        async with DOWNLOAD_SEMAPHORE:
            result = await download_video(url)

        # 5) Videoni yuborish
        await bot.send_chat_action(message.chat.id, ChatAction.UPLOAD_VIDEO)
        url_key = remember_url(url)
        caption = f"🎬 <b>{html.escape(result.title[:200])}</b>\n\n📥 Platforma: {platform}"

        await message.answer_video(
            video=FSInputFile(result.file_path),
            caption=caption,
            duration=result.duration,
            width=result.width,
            height=result.height,
            supports_streaming=True,
            reply_markup=audio_keyboard(url_key),
        )

        # 6) Holat xabarini o'chiramiz
        await safe_delete(status_msg)

    except FileTooLargeError:
        await safe_delete(status_msg)
        await message.answer(TOO_LARGE_TEXT)

    except Exception as exc:  # noqa: BLE001
        # Yaroqsiz havola, o'chirilgan video, tarmoq xatosi va h.k.
        logger.warning("Video yuklashda xatolik (%s): %s", url, exc)
        await safe_delete(status_msg)
        try:
            await message.answer(ERROR_TEXT)
        except Exception as send_exc:  # noqa: BLE001
            logger.error("Xato xabarini yuborib bo'lmadi: %s", send_exc)

    finally:
        # 7) Vaqtinchalik fayllarni har doim o'chiramiz (cleanup)
        if result:
            cleanup_folder(result.folder)

@router.callback_query(F.data.startswith("audio:"))
async def handle_audio(callback: CallbackQuery, bot: Bot) -> None:
    """Videoning musiqasini MP3 ko'rinishida ajratib yuboradi."""
    message = callback.message
    if not isinstance(message, Message):
        await callback.answer()
        return

    url_key = (callback.data or "").split(":", 1)[1]
    url = URL_CACHE.get(url_key)

    if not url:
        await callback.answer(EXPIRED_LINK_TEXT, show_alert=True)
        return

    # Tugma "aylanib" qolmasligi uchun darhol javob beramiz
    await callback.answer("🎵 Musiqa ajratilmoqda...")

    status_msg: Optional[Message] = None
    result = None

    try:
        status_msg = await message.answer(AUDIO_DOWNLOADING_TEXT)
        await bot.send_chat_action(message.chat.id, ChatAction.UPLOAD_VOICE)

        async with DOWNLOAD_SEMAPHORE:
            result = await download_audio(url)

        await bot.send_chat_action(message.chat.id, ChatAction.UPLOAD_VOICE)
        await message.answer_audio(
            audio=FSInputFile(
                result.file_path,
                filename=f"{safe_filename(result.title)}.mp3",
            ),
            title=result.title[:64],
            performer=(result.uploader or "")[:64] or None,
            duration=result.duration,
            caption="🎧 Musiqa ajratib olindi",
        )

        await safe_delete(status_msg)

    except FileTooLargeError:
        await safe_delete(status_msg)
        await message.answer(TOO_LARGE_TEXT)

    except Exception as exc:  # noqa: BLE001
        logger.warning("Audio ajratishda xatolik (%s): %s", url, exc)
        await safe_delete(status_msg)
        try:
            await message.answer(ERROR_TEXT)
        except Exception as send_exc:  # noqa: BLE001
            logger.error("Xato xabarini yuborib bo'lmadi: %s", send_exc)

    finally:
        if result:
            cleanup_folder(result.folder)
downloader.py
"""
downloader.py — havolani aniqlash va yt-dlp orqali video/audio yuklab olish.

Barcha og'ir (bloklovchi) ishlar alohida oqimda (thread) bajariladi,
shuning uchun bot boshqa foydalanuvchilarga javob berishda davom etadi.
"""

import asyncio
import logging
import re
import shutil
import uuid
from dataclasses import dataclass
from pathlib import Path
from typing import Optional
from urllib.parse import urlparse

import yt_dlp
from yt_dlp.utils import match_filter_func

from config import (
    COOKIES_FILE,
    DOWNLOAD_DIR,
    MAX_DURATION_SECONDS,
    MAX_FILE_SIZE_MB,
)

logger = logging.getLogger(__name__)

# ---------------------------------------------------------------------------
# Xatolik turlari
# ---------------------------------------------------------------------------


class DownloaderError(Exception):
    """Yuklab olishdagi umumiy xatolik."""


class FileTooLargeError(DownloaderError):
    """Fayl Telegram uchun juda katta."""


# ---------------------------------------------------------------------------
# Havolani aniqlash
# ---------------------------------------------------------------------------

# Matn ichidan havolani topish uchun regex
URL_REGEX = re.compile(r"https?://[^\s<>\"']+", re.IGNORECASE)

# Platforma nomi -> domen (host) regexi
PLATFORM_HOSTS: dict[str, re.Pattern] = {
    "Instagram": re.compile(r"(^|\.)(instagram\.com|instagr\.am)$"),
    "YouTube": re.compile(r"(^|\.)(youtube\.com|youtu\.be|youtube-nocookie\.com)$"),
    "TikTok": re.compile(r"(^|\.)tiktok\.com$"),
    # Pinterest turli mamlakat domenlarida bo'ladi (pinterest.com, pinterest.co.uk ...)
    "Pinterest": re.compile(r"(^|\.)(pinterest\.[a-z.]{2,}|pin\.it)$"),
}


def extract_url(text: str) -> Optional[str]:
    """Matn ichidan birinchi havolani ajratib oladi."""
    match = URL_REGEX.search(text or "")
    if not match:
        return None
    # Havola oxiridagi ortiqcha belgilarni (nuqta, vergul, qavs) olib tashlaymiz
    return match.group(0).rstrip(".,;:!?)]}")


def detect_platform(url: str) -> Optional[str]:
    """Havola qaysi platformaga tegishli ekanini aniqlaydi."""
    try:
        host = (urlparse(url).hostname or "").lower()
    except ValueError:
        return None

    for platform, pattern in PLATFORM_HOSTS.items():
        if pattern.search(host):
            return platform
    return None


# ---------------------------------------------------------------------------
# Natija strukturasi
# ---------------------------------------------------------------------------


@dataclass
class DownloadResult:
    """Yuklab olingan faylning ma'lumotlari."""

    folder: Path  # shu yuklash uchun yaratilgan alohida papka
    file_path: Path  # yuklangan fayl yo'li
    title: str
    uploader: Optional[str]
    duration: Optional[int]
    width: Optional[int]
    height: Optional[int]


# ---------------------------------------------------------------------------
# Yordamchi funksiyalar
# ---------------------------------------------------------------------------


def cleanup_folder(folder: Optional[Path]) -> None:
    """Vaqtinchalik papkani butunlay o'chiradi (xatolik bo'lsa ham jim turadi)."""
    if not folder:
        return
    try:
        shutil.rmtree(folder, ignore_errors=True)
    except Exception as exc:  # noqa: BLE001
        logger.warning("Papkani o'chirib bo'lmadi %s: %s", folder, exc)


def cleanup_all_downloads() -> None:
    """Bot ishga tushganda downloads/ papkasidagi eski qoldiqlarni tozalaydi."""
    DOWNLOAD_DIR.mkdir(parents=True, exist_ok=True)
    for item in DOWNLOAD_DIR.iterdir():
        try:
            if item.is_dir():
                shutil.rmtree(item, ignore_errors=True)
            else:
                item.unlink(missing_ok=True)
        except Exception as exc:  # noqa: BLE001
            logger.warning("Eski faylni o'chirib bo'lmadi %s: %s", item, exc)


def _find_media_file(folder: Path) -> Optional[Path]:
    """Papkadagi eng katta tayyor media faylni topadi (.part va h.k. larni o'tkazib yuboradi)."""
    ignored_suffixes = {".part", ".ytdl", ".temp", ".json", ".jpg", ".jpeg", ".png", ".webp", ".vtt", ".srt"}
    candidates = [
        p for p in folder.iterdir()
        if p.is_file() and p.suffix.lower() not in ignored_suffixes
    ]
    if not candidates:
        return None
    return max(candidates, key=lambda p: p.stat().st_size)


def _common_options(folder: Path) -> dict:
    """Video va audio uchun umumiy yt-dlp sozlamalari."""
    opts: dict = {
        "outtmpl": str(folder / "media.%(ext)s"),
        "noplaylist": True,  # faqat bitta videoni yuklaymiz
        "quiet": True,
        "no_warnings": True,
        "retries": 3,
        "socket_timeout": 30,
        "restrictfilenames": True,
        # Juda uzun videolarni o'tkazib yuboramiz ("<=?" — davomiylik noma'lum bo'lsa ham o'tkazadi)
        "match_filter": match_filter_func(f"duration <=? {MAX_DURATION_SECONDS}"),
    }
    if COOKIES_FILE:
        opts["cookiefile"] = COOKIES_FILE
    return opts


def _video_options(folder: Path) -> dict:
    """Video yuklash sozlamalari: Telegram'ga mos H.264/MP4, hajm chegarasi ichida eng sifatli."""
    size = f"{MAX_FILE_SIZE_MB}M"
    opts = _common_options(folder)
    opts.update(
        {
            # Format tanlash (chapdan o'ngga, birinchi mos kelgani olinadi):
            # 1) H.264 video + M4A audio (hajm chegarasi ichida)
            # 2) tayyor MP4 (hajm chegarasi ichida)
            # 3) istalgan format (hajm chegarasi ichida)
            # 4) oxirgi chora — eng yaxshi mavjud format
            # TikTok'da yt-dlp standart holda suv belgisiz formatni tanlaydi.
            "format": (
                f"bv*[vcodec^=avc1][filesize<?{size}][filesize_approx<?{size}]+ba[ext=m4a]"
                f"/b[ext=mp4][filesize<?{size}][filesize_approx<?{size}]"
                f"/b[filesize<?{size}][filesize_approx<?{size}]"
                "/b"
            ),
            "merge_output_format": "mp4",  # ffmpeg kerak
        }
    )
    return opts


def _audio_options(folder: Path) -> dict:
    """Audio yuklash sozlamalari: eng yaxshi audio -> MP3 (ffmpeg kerak)."""
    opts = _common_options(folder)
    opts.update(
        {
            "format": "bestaudio/best",
            "postprocessors": [
                {
                    "key": "FFmpegExtractAudio",
                    "preferredcodec": "mp3",
                    "preferredquality": "192",
                }
            ],
        }
    )
    return opts


# ---------------------------------------------------------------------------
# Asosiy yuklash mantig'i (sinxron)
# ---------------------------------------------------------------------------


def _download_sync(url: str, audio_only: bool) -> DownloadResult:
    """yt-dlp yordamida faylni yuklab oladi. Alohida oqimda chaqiriladi."""
    # Har bir yuklash uchun noyob papka — fayllar aralashib ketmaydi
    folder = DOWNLOAD_DIR / uuid.uuid4().hex
    folder.mkdir(parents=True, exist_ok=True)

    try:
        options = _audio_options(folder) if audio_only else _video_options(folder)

        with yt_dlp.YoutubeDL(options) as ydl:
            info = ydl.extract_info(url, download=True)

        if not info:
            raise DownloaderError("Havoladan ma'lumot olinmadi.")

        # Agar natija bir nechta elementdan iborat bo'lsa (masalan, Instagram karusel),
        # birinchisini olamiz
        if "entries" in info:
            entries = [e for e in (info.get("entries") or []) if e]
            if not entries:
                raise DownloaderError("Havolada video topilmadi.")
            info = entries[0]

        file_path = _find_media_file(folder)
        if not file_path:
            # Fayl yuklanmagan: davomiylik yoki hajm chegarasidan oshgan bo'lishi mumkin
            raise DownloaderError("Fayl yuklanmadi (chegaradan oshgan yoki mavjud emas).")

        # Telegram chegarasini tekshiramiz
        size_mb = file_path.stat().st_size / (1024 * 1024)
        if size_mb > MAX_FILE_SIZE_MB:
            raise FileTooLargeError(f"Fayl hajmi {size_mb:.1f} MB.")

        duration = info.get("duration")
        return DownloadResult(
            folder=folder,
            file_path=file_path,
            title=(info.get("title") or "Media").strip(),
            uploader=info.get("uploader") or info.get("channel"),
            duration=int(duration) if duration else None,
            width=info.get("width"),
            height=info.get("height"),
        )

    except Exception:
        # Xatolik bo'lsa — papkani darhol tozalab, xatoni yuqoriga uzatamiz
        cleanup_folder(folder)
        raise


# ---------------------------------------------------------------------------
# Asinxron interfeys (handlers.py shularni chaqiradi)
# ---------------------------------------------------------------------------


async def download_video(url: str) -> DownloadResult:
    """Videoni yuklab oladi (event loop'ni bloklamaydi)."""
    return await asyncio.to_thread(_download_sync, url, False)


async def download_audio(url: str) -> DownloadResult:
    """Videodan musiqani MP3 ko'rinishida ajratib oladi."""
    return await asyncio.to_thread(_download_sync, url, True)
